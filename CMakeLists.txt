cmake_minimum_required(VERSION 3.20)
project(mimicdb LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(MIMICDB_BUILD_TESTS "Build MimicDB tests" ON)
option(MIMICDB_BUILD_BENCHMARKS "Build MimicDB benchmarks" OFF)
option(MIMICDB_ENABLE_LTO "Enable link-time optimization" OFF)
option(MIMICDB_NATIVE_ARCH "Enable -march/-mtune=native" OFF)
option(MIMICRAG_BUILD_SERVER "Build the native MimicRAG HTTP server" ON)
option(MIMICRAG_ENABLE_LLAMA "Link the llama.cpp submodule for in-process local embeddings" ON)
set(MIMICRAG_LLAMA_GPU "auto" CACHE STRING "llama.cpp backend: auto, cuda, vulkan, metal, or cpu")
set_property(CACHE MIMICRAG_LLAMA_GPU PROPERTY STRINGS auto cuda vulkan metal cpu)
set(MIMICDB_PGO "off" CACHE STRING "PGO mode: off, generate, use")
set_property(CACHE MIMICDB_PGO PROPERTY STRINGS off generate use)

if(MIMICDB_ENABLE_LTO)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT lto_supported OUTPUT lto_error)
    if(lto_supported)
        set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
    else()
        message(WARNING "LTO requested but not supported: ${lto_error}")
    endif()
endif()

add_library(mimicdb_build_flags INTERFACE)

if(MIMICDB_NATIVE_ARCH)
    target_compile_options(mimicdb_build_flags INTERFACE -march=native -mtune=native)
endif()

if(MIMICDB_PGO STREQUAL "generate")
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(mimicdb_build_flags INTERFACE -fprofile-generate)
        target_link_options(mimicdb_build_flags INTERFACE -fprofile-generate)
    endif()
elseif(MIMICDB_PGO STREQUAL "use")
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(mimicdb_build_flags INTERFACE -fprofile-use -fprofile-correction)
        target_link_options(mimicdb_build_flags INTERFACE -fprofile-use -fprofile-correction)
    endif()
endif()

add_subdirectory(engine)
add_subdirectory(server)
if(MIMICDB_BUILD_TESTS)
    enable_testing()
endif()
if(MIMICRAG_BUILD_SERVER)
    add_subdirectory(rag_cpp)
endif()
add_subdirectory(api_cpp)

if(MIMICDB_BUILD_TESTS)
    add_subdirectory(tests)
endif()

if(MIMICDB_BUILD_BENCHMARKS)
    add_executable(mimicdb_bench_scan benchmarks/bench_scan.cpp)
    target_link_libraries(mimicdb_bench_scan PRIVATE mimicdb_engine mimicdb_build_flags)
    add_executable(mimicdb_bench_vector benchmarks/bench_vector.cpp)
    target_link_libraries(mimicdb_bench_vector PRIVATE mimicdb_engine mimicdb_build_flags)
    add_executable(mimicdb_bench_vector_file benchmarks/bench_vector_file.cpp)
    target_link_libraries(mimicdb_bench_vector_file PRIVATE mimicdb_engine mimicdb_build_flags)
endif()
